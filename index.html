Decisiones de seguridad

🔐

IDOR corregido

usuario_id siempre se extrae del token JWT, nunca de path params ni del body. El usuario no puede leer datos de otros.

📋

Whitelist de emails

El registro está bloqueado para emails no autorizados. Solo el superadmin puede agregar emails a la lista blanca.

🛡️

require_superadmin

Dependencia FastAPI que compone sobre get_current_user y protege todos los endpoints /admin/* con un solo Depends().

🔑

bcrypt + JWT HS256

Passwords hasheados con passlib[bcrypt]. Tokens con 7 días de expiración, firmados con SECRET_KEY en variables de entorno.

snippets/backend/classifier_excerpt.py — motor de clasificación local

def _extraer_monto(texto: str) -> Optional[float]:
    """Extrae el primer monto numérico encontrado en el texto."""
    m = _PATRON_MONTO.search(texto)
    if not m:
        return None
    entero = m.group(1).replace(".", "")
    decimal = m.group(2) if m.group(2) else "0"
    valor = float(f"{entero}.{decimal}")
    return valor if valor > 0 else None


def clasificar(texto: str) -> Optional[ClasificacionLocal]:
    """
    Intenta clasificar el texto usando solo reglas locales.
    Retorna None si el texto es ambiguo → delega a la IA.
    """
    texto_limpio = texto.lower().strip()

    monto = _extraer_monto(texto_limpio)
    if monto is None:
        return None  # Sin monto claro → delegar a IA

    tipo = _detectar_tipo(texto_limpio)
    categoria = _detectar_categoria(texto_limpio, tipo)

    # Texto elaborado sin categoría → delegar a IA
    if categoria is None and len(texto_limpio.split()) > 3:
        return None

    categoria_final = categoria or "Varios"
    pilar = CATEGORIA_A_PILAR.get(categoria_final, "estilo_vida")

    return ClasificacionLocal(
        monto=monto, tipo=tipo, pilar=pilar,
        categoria=categoria_final,
        descripcion=texto.strip().capitalize(),
    )

snippets/backend/security_excerpt.py — JWT + dependencias FastAPI

bearer_scheme = HTTPBearer()

def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(bearer_scheme),
    db: Session = Depends(get_db),
) -> Usuario:
    token = credentials.credentials
    try:
        payload = jwt.decode(
            token, settings.SECRET_KEY,
            algorithms=[settings.ALGORITHM]
        )
        user_id: int = int(payload.get("sub"))
    except (JWTError, TypeError, ValueError):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Token inválido o expirado",
        )

    user = db.query(Usuario).filter(Usuario.id == user_id).first()
    if not user:
        raise HTTPException(status_code=404, detail="Usuario no encontrado")
    if not user.is_active:
        raise HTTPException(status_code=403, detail="Cuenta desactivada")
    return user


def require_superadmin(
    current_user: Usuario = Depends(get_current_user)
) -> Usuario:
    """Guard para todos los endpoints /admin/*."""
    if current_user.role != "superadmin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Acceso restringido a superadministradores",
        )
    return current_user

snippets/backend/rate_limiting_excerpt.py — protección contra fuerza bruta en login

from slowapi import Limiter
from slowapi.util import get_remote_address

limiter = Limiter(key_func=get_remote_address)
router = APIRouter(prefix="/auth", tags=["Auth"])


@router.post("/login")
@limiter.limit("10/minute")
async def login(
    request: Request,
    credentials: LoginPayload,
    db: Session = Depends(get_db),
):
    user = authenticate_user(db, credentials.email, credentials.password)
    if not user:
        raise HTTPException(status_code=401, detail="Credenciales inválidas")
    token = create_access_token(subject=user.id)
    return {"access_token": token, "token_type": "bearer", "user": user}

export const AuthProvider: FC<{ children: ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [token, setToken] = useState<string | null>(
    localStorage.getItem("token")
  );
  const [isLoading, setIsLoading] = useState(true);

  // Derivados — sin estado extra ni sincronización manual.
  const isAuthenticated = !!token;
  const isSuperAdmin = user?.role === "superadmin";

  // Re-valida la sesión al montar. Si el backend devuelve 401 → logout.
  useEffect(() => {
    const initAuth = async () => {
      if (token) {
        try {
          const userData = await authService.getMe();
          setUser(userData);
        } catch {
          logout();
        }
      }
      setIsLoading(false);
    };
    initAuth();
  }, [token]);

  // ...login, register, logout expuestos en el contexto
};

snippets/frontend/AdminRoute_excerpt.tsx — guard de ruta React Router v7

export function AdminRoute() {
  const { isAuthenticated, isSuperAdmin, isLoading } = useAuth();

  if (isLoading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="w-8 h-8 border-4 border-indigo-500
                        border-t-transparent rounded-full animate-spin" />
      </div>
    );
  }

  // Doble guard: auth + rol.
  if (!isAuthenticated) return <Navigate to="/login" replace />;
  if (!isSuperAdmin)    return <Navigate to="/dashboard" replace />;

  return <Outlet />;
}
